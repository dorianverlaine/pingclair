# 現代化設定範例 | Modern Configuration
server "example.com" {
    listen: "0.0.0.0:443";
    tls auto;
    route {
        match path("/api/*") => {
            proxy { to "localhost:8080"; }
        }
        _ => {
            file_server "/var/www/html";
        }
    }
}
